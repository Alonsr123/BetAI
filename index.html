export default async function handler(req, res) { try { const API_KEY = process.env.API_FOOTBALL_KEY; if (!API_KEY) { return res.status(500).json({ error: "No existe API_FOOTBALL_KEY en las variables de Vercel" }); } // Fecha solicitada desde BetAI. // Si no se proporciona, usamos la fecha actual en España. const requestedDate = req.query.date; let date = requestedDate; if (!date) { const now = new Date(); date = new Intl.DateTimeFormat("en-CA", { timeZone: "Europe/Madrid" }).format(now); } /* Competiciones que queremos en BetAI. Los IDs son los de API-Football. */ const leagues = [ // España { id: 140, name: "LaLiga", country: "España", flag: "🇪🇸" }, { id: 141, name: "Segunda División", country: "España", flag: "🇪🇸" }, // Inglaterra { id: 39, name: "Premier League", country: "Inglaterra", flag: "🏴" }, { id: 40, name: "Championship", country: "Inglaterra", flag: "🏴" }, // Italia { id: 135, name: "Serie A", country: "Italia", flag: "🇮🇹" }, { id: 136, name: "Serie B", country: "Italia", flag: "🇮🇹" }, // Alemania { id: 78, name: "Bundesliga", country: "Alemania", flag: "🇩🇪" }, { id: 79, name: "2. Bundesliga", country: "Alemania", flag: "🇩🇪" }, // Francia { id: 61, name: "Ligue 1", country: "Francia", flag: "🇫🇷" }, { id: 62, name: "Ligue 2", country: "Francia", flag: "🇫🇷" }, // Escandinavia { id: 113, name: "Allsvenskan", country: "Suecia", flag: "🇸🇪" }, { id: 103, name: "Eliteserien", country: "Noruega", flag: "🇳🇴" }, { id: 119, name: "Superliga", country: "Dinamarca", flag: "🇩🇰" }, { id: 244, name: "Veikkausliiga", country: "Finlandia", flag: "🇫🇮" }, { id: 166, name: "Besta deild", country: "Islandia", flag: "🇮🇸" }, // Europa { id: 2, name: "Champions League", country: "Europa", flag: "🏆" } ]; /* Consultamos las competiciones en paralelo. Promise.allSettled permite que una liga con problemas no bloquee todas las demás. */ const requests = leagues.map(async (league) => { const url = "https://v3.football.api-sports.io/fixtures" + "?league=" + league.id + "&date=" + encodeURIComponent(date) + "&timezone=Europe/Madrid"; const response = await fetch(url, { headers: { "x-apisports-key": API_KEY } }); const data = await response.json(); return { league, http: response.status, data }; }); const results = await Promise.allSettled(requests); const partidos = []; const errores = []; results.forEach((result) => { if (result.status === "rejected") { errores.push({ error: result.reason?.message || "Error desconocido" }); return; } const item = result.value; const league = item.league; const data = item.data; if (item.http !== 200) { errores.push({ liga: league.name, http: item.http, error: data?.errors || {} }); return; } if (data?.errors && Object.keys(data.errors).length > 0) { errores.push({ liga: league.name, error: data.errors }); } /* Convertimos la respuesta de API-Football a un formato sencillo para BetAI. */ (data.response || []).forEach((fixture) => { partidos.push({ id: fixture.fixture.id, fecha: fixture.fixture.date, estado: fixture.fixture.status.short, estadoLargo: fixture.fixture.status.long, minuto: fixture.fixture.status.elapsed, local: fixture.teams.home.name, visitante: fixture.teams.away.name, logoLocal: fixture.teams.home.logo, logoVisitante: fixture.teams.away.logo, golesLocal: fixture.goals.home, golesVisitante: fixture.goals.away, liga: league.name, ligaId: league.id, pais: league.country, bandera: league.flag, ronda: fixture.league.round, temporada: fixture.league.season }); }); }); /* Ordenamos cronológicamente. */ partidos.sort((a, b) => { return new Date(a.fecha) - new Date(b.fecha); }); return res.status(200).json({ fecha: date, total: partidos.length, competiciones: leagues, errores, partidos }); } catch (error) { return res.status(500).json({ error: "Error interno del servidor", detalle: error.message }); } }
